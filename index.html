// sw.js const APP_VERSION = 'v1.0.0'; const CACHE_NAME = `growth-app-${APP_VERSION}`; // 메인 스크립트에서 버전 요청 시 응답 self.addEventListener('message', (event) => { if (event.data && event.data.type === 'GET_VERSION') { if (event.ports && event.ports[0]) { event.ports[0].postMessage({ version: APP_VERSION }); } } }); // 설치 및 캐싱 로직 self.addEventListener('install', (event) => { self.skipWaiting(); }); self.addEventListener('activate', (event) => { event.waitUntil( caches.keys().then((keys) => Promise.all( keys.map((key) => { if (key !== CACHE_NAME) return caches.delete(key); }) ) ) ); self.clients.claim(); });
